# api/requirements.txt
# Список всех Python пакетов которые нужны нашему API.
# pip установит их все при сборке контейнера.

# Веб-фреймворк — принимает запросы, отдаёт ответы
fastapi

# Веб-сервер который запускает FastAPI
uvicorn

# ORM — работа с PostgreSQL через Python классы вместо SQL
sqlalchemy

# Драйвер для подключения Python к PostgreSQL
psycopg2-binary

# Нужен чтобы FastAPI мог принимать файлы (наши MRI снимки)
python-multipart

# Глубокое обучение — загрузка и запуск нашей модели
torch
torchvision

# Работа с изображениями — открыть, изменить размер, конвертировать
Pillow
